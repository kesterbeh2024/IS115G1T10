git clone https://github.com/kesterbeh2024/IS115G1T10.git
cd IS115G1T10

# 1. Run Baseline FIFO Policy for Day 1

python -m simulator.simulator --policy solution.policies.baseline --data data --day day_001 --output results/baseline/day_001

# 2. Run Performance Monitor on Baseline Log

python -m solution.performance_monitor --data data --day day_001 --log results/baseline/day_001/log.txt

# 3. Run Custom Policy for Day 1

python -m simulator.simulator --policy solution.policies.heuristic_best --data data --day day_001 --output results/heuristic_best/day_001

# 4. Run Performance Monitor on Custom Policy Log

python -m solution.performance_monitor --data data --day day_001 --log results/heuristic_best/day_001/log.txt
